```typescript
export function applyPostMatchInjuries(team: Team, chance = 0.15, rng: () => number = Math.random): { team: Team; injuries: InjuryReport[] } {
  if (rng() >= chance) return { team, injuries: [] };
  const starters = team.squad.slice(0, 11).filter(p => !p.injuredRounds);
  if (starters.length === 0) return { team, injuries: [] };
  const victim = starters[Math.floor(rng() * starters.length)];
  const rounds = 1 + Math.floor(rng() * 3); // 1-3
  const newSquad = team.squad.map(p => p.id === victim.id ? { ...p, injuredRounds: rounds } : p);
  return { team: { ...team, squad: newSquad }, injuries: [{ player: { id: victim.id, name: victim.name, number: victim.number }, rounds }] };
}
```

```typescript
export function generatePlayer(id: string, name: string, number: number, pos: Player['position'], ovr: number): Player {
  const base = ovr - 7;
  const spread = Math.min(10, Math.floor(ovr / 8));
  const attr = () => Math.max(1, Math.min(99, base + rand(-spread, spread)));
  const boost = (n: number) => Math.max(1, Math.min(99, attr() + n));
  return {
    id, name, number, position: pos, currentFatigue: 0,
    attributes: {
      speed: pos === 'FWD' || pos === 'DEF' ? boost(8) : attr(),
      stamina: boost(2),
      passing: pos === 'MID' ? boost(10) : attr(),
      finishing: pos === 'FWD' ? boost(12) : pos === 'MID' ? boost(4) : attr(),
      tackling: pos === 'DEF' ? boost(12) : pos === 'MID' ? boost(4) : attr(),
      dribbling: pos === 'FWD' || pos === 'MID' ? boost(6) : attr(),
      vision: pos === 'MID' ? boost(8) : pos === 'FWD' ? boost(4) : attr(),
      positioning: pos === 'GK' ? boost(14) : boost(4),
    },
  };
}
```

```typescript
const WEIGHTS: Record<string, Record<string, number>> = {
  GK:  { speed: 0, stamina: 1.0, passing: 1.5, finishing: 0, tackling: 0, dribbling: 0, vision: 2.0, positioning: 4.0 },
  DEF: { speed: 1.8, stamina: 1.2, passing: 1.0, finishing: 0, tackling: 3.0, dribbling: 0, vision: 0, positioning: 2.5 },
  MID: { speed: 1.0, stamina: 1.2, passing: 2.5, finishing: 0, tackling: 0.8, dribbling: 1.8, vision: 2.5, positioning: 0 },
  FWD: { speed: 2.5, stamina: 0, passing: 0.8, finishing: 3.5, dribbling: 2.2, vision: 0, tackling: 0, positioning: 1.2 },
};
export function playerOvr(p: Player): number {
  const w = WEIGHTS[p.position] ?? WEIGHTS['MID'];
  const total = TOTALS[p.position] ?? TOTALS['MID'];
  const a = p.attributes;
  const sum = a.speed * (w.speed ?? 0) + a.stamina * (w.stamina ?? 0) + a.passing * (w.passing ?? 0)
    + a.finishing * (w.finishing ?? 0) + a.tackling * (w.tackling ?? 0) + a.dribbling * (w.dribbling ?? 0)
    + a.vision * (w.vision ?? 0) + a.positioning * (w.positioning ?? 0);
  return Math.round(sum / total);
}
```

**Trade CPU-decision heuristic** — already fully drafted in RESEARCH.md Code Examples, grounded in `randomizeCPUTactics`/`applyPostMatchInjuries` style — copy directly:
```typescript
import { playerOvr } from '../utils/playerOvr';
import type { Player } from './types';

export function evaluateTradeOffer(
  offeredPlayer: Player, requestedPlayer: Player, cpuSquad: Player[],
  managerReputation: number, rng: () => number = Math.random,
): { accepted: boolean; reason: string } {
  const offeredOvr = playerOvr(offeredPlayer);
  const requestedOvr = playerOvr(requestedPlayer);
  const ovrDelta = offeredOvr - requestedOvr;
  const samePositionCount = cpuSquad.filter(p => p.position === requestedPlayer.position).length;
  const surplus = samePositionCount > 3;
  let acceptChance = 0.5 + ovrDelta * 0.04;
  if (surplus) acceptChance += 0.15;
  acceptChance += Math.min(0.2, managerReputation * 0.002);
  acceptChance = Math.max(0.02, Math.min(0.95, acceptChance));
  return { accepted: rng() < acceptChance, reason: ovrDelta >= 0 ? 'value-favorable' : 'value-unfavorable' };
}
```

```typescript
export function usePersistedState<T>(
  key: string, initial: T | (() => T), opts?: { onPersist?: (value: T) => void }
): [T, React.Dispatch<React.SetStateAction<T>>] {
  const readFromStorage = (k: string): T => {
    try {
      const stored = localStorage.getItem(k);
      if (stored !== null) return JSON.parse(stored) as T;
    } catch { /* noop */ }
    return typeof initial === 'function' ? (initial as () => T)() : initial;
  };
  const [value, setValue] = useState<T>(() => readFromStorage(key));
  // ... key-swap reset on render (lines 28-32), persist-on-change effect with try/catch (lines 36-48)
  return [value, setValue];
}
```
